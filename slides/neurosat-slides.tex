% ==================================================================== problem
\begin{frame}{\T{The SAT problem}{Il problema SAT}}
  \begin{itemize}
    \item \T{SAT: given a Boolean formula in conjunctive normal form}%
          {SAT: data una formula booleana in forma normale congiuntiva}
          $\varphi=\bigwedge_j \big(\bigvee_{\ell \in C_j} \ell\big)$,
          \T{is there an assignment of the variables that satisfies all its clauses? It
          was the first problem proven NP-complete, and it is the engine of verification,
          planning and synthesis.}%
          {esiste un assegnamento delle variabili che ne soddisfa tutte le clausole? \`E
          stato il primo problema dimostrato NP-completo, ed \`e il motore di verifica,
          pianificazione e sintesi.}
    \item \T{Modern solvers are CDCL: they \emph{decide} a variable, \emph{propagate}, and on
          a \emph{conflict} learn a clause and backjump.}%
          {I solutori moderni sono CDCL: \emph{decidono} una variabile, \emph{propagano}, e a
          ogni \emph{conflitto} imparano una clausola e fanno backjump.}
    \item \T{The branching heuristic (which variable to decide next, usually VSIDS) drives
          the performance, but its activity counters carry little information at the start
          of the search.}%
          {L'euristica di branching (quale variabile decidere, di solito VSIDS) determina le
          prestazioni, ma all'inizio della ricerca i suoi contatori di attivit\`a dicono poco.}
    \item \T{This project: learn the branching heuristic with two neuro-symbolic methods,
          port both to a current PyTorch stack, check the port against the released
          checkpoints, and measure where graph attention helps.}%
          {Questo progetto: imparare l'euristica di branching con due metodi neuro-simbolici,
          portarli entrambi a uno stack PyTorch attuale, verificare il port sui checkpoint
          pubblicati, e misurare dove l'attenzione su grafo aiuta.}
  \end{itemize}
\end{frame}

\begin{frame}{\T{The CNF as a graph}{La CNF come grafo}}
  \begin{columns}[t]
    \begin{column}{0.6\textwidth}
      \T{A CNF is naturally a bipartite graph:}{Una CNF \`e in modo naturale un grafo bipartito:}
      \begin{itemize}
        \item \T{a node for every variable (or literal) and every clause;}%
              {un nodo per ogni variabile (o letterale) e per ogni clausola;}
        \item \T{an edge between a variable and each clause in which it occurs, with the
              polarity as edge feature (in the view by literals, also an edge between $x$
              and $\bar x$).}%
              {un arco fra una variabile e ogni clausola in cui compare, con la polarit\`a come
              feature dell'arco (nella vista per letterali, anche un arco fra $x$ e $\bar x$).}
      \end{itemize}
      \vspace{2mm}
      \T{The representation is invariant under permutations of variables and clauses, and
      one network applies to formulas of any size, which the fixed-size CNN cannot do. It is
      the representation of NeuroSAT and of Graph-Q-SAT.}%
      {La rappresentazione \`e invariante per permutazioni di variabili e clausole, e una
      sola rete si applica a formule di qualunque dimensione, cosa che la CNN a dimensione
      fissa non pu\`o fare. \`E la rappresentazione di NeuroSAT e di Graph-Q-SAT.}
    \end{column}
    \begin{column}{0.4\textwidth}
      \centering
      \footnotesize
      $\varphi=(x_1\vee \bar x_2)\wedge(\bar x_1 \vee x_2 \vee x_3)$\\[3mm]
      \begin{tabular}{c@{\hspace{6mm}}c}
        \T{variables}{variabili} & \T{clauses}{clausole}\\\midrule
        $x_1$ & \\
        $x_2$ & $c_1$\\
        $x_3$ & $c_2$\\
      \end{tabular}\\[3mm]
      {\scriptsize \T{the edges carry the polarity $\{+,-\}$ of the literal.}%
      {gli archi portano la polarit\`a $\{+,-\}$ del letterale.}}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Message Passing Neural Networks (MPNN)}
  \T{A general framework for learning on graphs (Gilmer et al., 2017). Each node $v$ holds a
  state $h_v^{t}$, and for $T$ rounds it exchanges messages with its neighbors $\mathcal N(v)$:}%
  {Uno schema generale per l'apprendimento su grafi (Gilmer et al., 2017). Ogni nodo $v$ ha uno
  stato $h_v^{t}$, e per $T$ round scambia messaggi con i suoi vicini $\mathcal N(v)$:}
  \begin{align*}
    \text{\T{message:}{messaggio:}}\quad & m_v^{t+1} = \sum_{w\in\mathcal N(v)} M_t\!\big(h_v^{t},\,h_w^{t},\,e_{vw}\big)\\[1mm]
    \text{\T{update:}{aggiornamento:}}\quad  & h_v^{t+1} = U_t\!\big(h_v^{t},\,m_v^{t+1}\big)\\[1mm]
    \text{\T{readout:}{lettura:}}\quad & \hat y = R\big(\{\,h_v^{T} : v\in G\,\}\big)
  \end{align*}
  \begin{itemize}
    \item \T{$M_t$ (message), $U_t$ (update, often a GRU or an LSTM) and $R$ (readout) are
          learned and shared across the nodes and the rounds.}%
          {$M_t$ (messaggio), $U_t$ (aggiornamento, spesso una GRU o una LSTM) e $R$ (lettura)
          sono appresi e condivisi fra i nodi e fra i round.}
    \item \T{Gated GNNs, GCN, GAT and the encode--process--decode network are all instances
          of this template.}%
          {Le GNN con gate, le GCN, le GAT e la rete encode--process--decode sono tutte
          istanze di questo schema.}
  \end{itemize}
\end{frame}

\begin{frame}{\T{NeuroSAT: a SAT solver from single-bit supervision}{NeuroSAT: un solutore SAT da una supervisione di un bit}}
  \begin{columns}[t]
    \begin{column}{0.54\textwidth}
      \footnotesize
      \T{Literal and clause embeddings}{Embedding di letterali e clausole}
      $L^{(t)}\!\in\!\mathbb{R}^{2n\times d}$, $C^{(t)}\!\in\!\mathbb{R}^{m\times d}$;
      \T{bipartite adjacency $M$; perceptrons}{adiacenza bipartita $M$; percettroni}
      $L_{\text{msg}},C_{\text{msg}},L_{\text{vote}}$; \T{LSTM updates}{aggiornamenti LSTM}
      $L_u,C_u$. \T{One round:}{Un round:}
      \begin{align*}
        (C^{(t+1)}\!,C_h^{(t+1)}) &\leftarrow C_u\big([\,C_h^{(t)},\,M^{\!\top} L_{\text{msg}}(L^{(t)})\,]\big)\\
        (L^{(t+1)}\!,L_h^{(t+1)}) &\leftarrow L_u\big([\,L_h^{(t)},\,F(L^{(t)}),\,M\,C_{\text{msg}}(C^{(t+1)})\,]\big)
      \end{align*}
      \T{after $T$ rounds, vote and average:}{dopo $T$ round, voto e media:}
      \[
        L_*^{(T)}\leftarrow L_{\text{vote}}(L^{(T)}),\qquad
        y^{(T)}\leftarrow \operatorname{mean}\big(L_*^{(T)}\big).
      \]
      \T{$F$ swaps each literal with its complement.}{$F$ scambia ogni letterale con il suo complemento.}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{itemize}
        \item \T{Trained only as a classifier of satisfiability (one bit per formula).}%
              {Addestrato solo come classificatore di soddisfacibilit\`a (un bit per formula).}
        \item \T{Yet it learns to find satisfying assignments, which appear in the votes of
              the literals.}%
              {Eppure impara a trovare assegnamenti che soddisfano la formula, che compaiono
              nei voti dei letterali.}
        \item \T{It established the graph view of a CNF, on which Graph-Q-SAT is built.}%
              {Ha stabilito la vista a grafo di una CNF, su cui \`e costruito Graph-Q-SAT.}
        \item \T{Limitation: a standalone solver, not competitive in running time with CDCL.}%
              {Limite: un solutore a s\'e, non competitivo nel tempo con CDCL.}
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{\T{Branching as a Markov Decision Process}{Il branching come processo decisionale di Markov}}
  \T{Graph-Q-SAT turns the CDCL search into an MDP $(\mathcal S,\mathcal A,P,r,\gamma)$:}%
  {Graph-Q-SAT trasforma la ricerca CDCL in un MDP $(\mathcal S,\mathcal A,P,r,\gamma)$:}
  \begin{itemize}
    \item \T{state $s$: the formula simplified by the partial assignment, with the learned
          clauses, as a bipartite graph;}%
          {stato $s$: la formula semplificata dall'assegnamento parziale, con le clausole
          imparate, come grafo bipartito;}
    \item \T{action $a$: an unassigned variable with a truth value, i.e., one decision;}%
          {azione $a$: una variabile libera con un valore di verit\`a, cio\`e una decisione;}
    \item \T{reward $r=-0.1$ at every non-terminal step and $0$ at termination, so that
          maximizing the return minimizes the number of decisions.}%
          {ricompensa $r=-0.1$ a ogni passo non terminale e $0$ alla fine, quindi
          massimizzare il ritorno minimizza il numero di decisioni.}
  \end{itemize}
  \vspace{1mm}
  \T{The goal is the action-value function}{L'obiettivo \`e la funzione di valore delle azioni}
  \[
    Q^\star(s,a)=\mathbb E\big[\,r + \gamma \max_{a'} Q^\star(s',a')\,\big]
    \qquad\text{(Bellman)},
  \]
  \T{and the heuristic is $\pi(s)=\arg\max_a Q^\star(s,a)$ over the variable nodes.}%
  {e l'euristica \`e $\pi(s)=\arg\max_a Q^\star(s,a)$ sui nodi variabile.}
\end{frame}

\begin{frame}{Deep Q-Learning (DQN)}
  \T{$Q^\star$ is approximated by a network $Q_\theta$, fitted by minimizing the
  temporal-difference error over a replay buffer $\mathcal D$:}%
  {$Q^\star$ \`e approssimata da una rete $Q_\theta$, adattata minimizzando l'errore di
  differenza temporale su un replay buffer $\mathcal D$:}
  \[
    \mathcal L(\theta)=\mathbb E_{(s,a,r,s')\sim\mathcal D}\!\left[
      \Big(\underbrace{r+\gamma\max_{a'}Q_{\theta^-}(s',a')}_{\text{TD target}}-Q_\theta(s,a)\Big)^{2}
    \right]
  \]
  \begin{itemize}
    \item \T{the target network $\theta^-$, a copy of $\theta$ refreshed every 10 updates,
          stabilizes the target;}%
          {la target network $\theta^-$, una copia di $\theta$ aggiornata ogni 10 passi,
          stabilizza il target;}
    \item \T{the replay buffer (20000 transitions, sampled uniformly) breaks the correlation
          between consecutive transitions;}%
          {il replay buffer (20000 transizioni, campionate in modo uniforme) rompe la
          correlazione fra transizioni consecutive;}
    \item \T{$\varepsilon$-greedy exploration: a random action with probability
          $\varepsilon$, otherwise $\arg\max_a Q_\theta$.}%
          {esplorazione $\varepsilon$-greedy: un'azione a caso con probabilit\`a
          $\varepsilon$, altrimenti $\arg\max_a Q_\theta$.}
  \end{itemize}
  \T{The number of actions changes along an episode (two per unassigned variable), and
  hence $Q_\theta$ is a graph network that scores each variable node.}%
  {Il numero di azioni cambia lungo un episodio (due per variabile libera), e quindi
  $Q_\theta$ \`e una rete su grafo che d\`a un punteggio a ogni nodo variabile.}
\end{frame}

\begin{frame}{\T{Model: the encode--process--decode graph network}{Modello: la rete encode--process--decode}}
  \begin{columns}[c]
    \begin{column}{0.4\textwidth}
      \footnotesize
      \T{A graph-network block (Battaglia et al.) updates the edges, then the nodes, then
      the global attribute:}{Un blocco graph network (Battaglia et al.) aggiorna gli archi,
      poi i nodi, poi l'attributo globale:}
      \begin{align*}
        e_{ij}' &= \phi^e(e_{ij}, v_i, v_j, u)\\
        v_i'    &= \phi^v\big(\textstyle\sum_{j\in\mathcal N_i} e_{ij}', v_i, u\big)\\
        u'      &= \phi^u(\bar e', \bar v', u)
      \end{align*}
      \T{$\phi$ are perceptrons; the sum is the aggregation $\rho$ of the figure. The
      encoder and the decoder act on each element independently; the core block is applied
      $M=4$ times with shared weights, and the decoder gives two $Q$-values per variable.}%
      {i $\phi$ sono percettroni; la somma \`e l'aggregazione $\rho$ della figura. Encoder e
      decoder agiscono su ogni elemento in modo indipendente; il blocco centrale \`e applicato
      $M=4$ volte con pesi condivisi, e il decoder d\`a due $Q$-value per variabile.}
    \end{column}
    \begin{column}{0.6\textwidth}
      \centering
      \includegraphics[width=\textwidth]{gqsat_epd.png}\\
      {\scriptsize encode--process--decode: $\mathrm{GN}_{enc}\to\mathrm{GN}_{core}{\times}M\to\mathrm{GN}_{dec}$.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{\T{Configuration and measure}{Configurazione e misura}}
  \begin{columns}[t]
    \begin{column}{0.5\textwidth}
      \T{The configuration of Kurin et al.:}{La configurazione di Kurin et al.:}
      \begin{itemize}
        \item \T{node features: 2-dimensional one-hot (variable or clause); edge features: the

        \item \T{embeddings of size 32 from the encoder, perceptrons with 64 hidden units in

              pi\`u 500 decisioni del modello per episodio (il cap).}
      \end{itemize}
    \end{column}

      \T{The measure, MRIR:}{La misura, MRIR:}
      \[
        \text{MRIR}=\operatorname{median}\frac{\#\text{iter}_{\text{MiniSat}}}{\#\text{iter}_{\text{\T{model}{modello}}}}
      \]
      \T{the Median Relative Iteration Reduction over the instances of a family: above $1$ the

      {la Median Relative Iteration Reduction sulle istanze di una famiglia: sopra $1$ il

      rimanenti.}
    \end{column}
  \end{columns}
\end{frame}

% ---------------------------------------------------------------- GAT
\begin{frame}{\T{Improvement idea: graph attention}{Idea di miglioramento: l'attenzione su grafo}}
  \begin{itemize}
    \item \T{In the core block a node sums the messages of all its neighbors with the same
          weight; on a random formula the clauses are statistically alike, but in a
          structured one they often play different roles (in graph coloring, ``at least one
          color'' for a node and ``different colors'' for the ends of an edge).}%
          {Nel blocco centrale un nodo somma i messaggi di tutti i suoi vicini con lo stesso
          peso; in una formula random le clausole sono statisticamente uguali, ma in una
          strutturata hanno spesso ruoli diversi (nel graph coloring, ``almeno un colore'' per
          un nodo e ``colori diversi'' per gli estremi di un arco).}
    \item \T{Idea: weight the neighbors by learned coefficients that depend on the edge
          embeddings, i.e., a Graph Attention Network (GAT) inside the core block.}%
          {Idea: pesare i vicini con coefficienti appresi che dipendono dagli embedding degli
          archi, cio\`e una Graph Attention Network (GAT) dentro il blocco centrale.}
    \item \T{In the terms of Battaglia et al., a non-local block inserted in the full
          graph-network block.}%
          {Nei termini di Battaglia et al., un blocco non locale inserito nel blocco graph
          network completo.}
  \end{itemize}
  \vfill
  {\footnotesize\color{gray} \href{https://arxiv.org/abs/1710.10903}{Veli\v{c}kovi\'c et al.,
  \emph{Graph Attention Networks}} (ICLR 2018).}
\end{frame}

\begin{frame}{Graph Attention Networks (GAT)}
  \T{For a node $i$ with neighbors $\mathcal N_i$ and a shared linear map $W$, the attention
  weights each message by a learned relevance:}%
  {Per un nodo $i$ con vicini $\mathcal N_i$ e una mappa lineare condivisa $W$, l'attenzione
  pesa ogni messaggio con una rilevanza appresa:}
  \begin{align*}
    \text{score:}\quad & \omega_{ij}=\text{LeakyReLU}\big(\mathbf w^{\!\top}[\,W v_i \,\|\, W v_j\,]\big)\\
    \text{\T{normalize:}{normalizzazione:}}\quad & \alpha_{ij}=\frac{\exp \omega_{ij}}{\sum_{j'\in\mathcal N_i}\exp \omega_{ij'}}\\
    \text{\T{aggregate:}{aggregazione:}}\quad & v_i^{+}=\textstyle\sum_{j\in\mathcal N_i}\alpha_{ij}\,W v_j
  \end{align*}
  \begin{itemize}
    \item \T{multi-head: $K$ independent attentions, concatenated or averaged;}%
          {multi-head: $K$ attenzioni indipendenti, concatenate o mediate;}
    \item \T{edge-featured: the score also reads the embedding of the edge, which carries
          the polarity of the literal.}%
          {con le feature degli archi: il punteggio legge anche l'embedding dell'arco, che
          porta la polarit\`a del letterale.}
  \end{itemize}
\end{frame}

\begin{frame}{Monte Carlo Tree Search (MCTS)}
  \T{Solving is treated as a game: a node is a partial assignment, a move a branching
  literal. From the root, four steps are repeated:}%
  {La risoluzione \`e trattata come un gioco: un nodo \`e un assegnamento parziale, una mossa un
  letterale di branching. Dalla radice si ripetono quattro passi:}
  \begin{enumerate}
    \item \T{select the child that maximizes the PUCT score}{selezione del figlio che massimizza il punteggio PUCT}
      \[
        a_t=\arg\max_a\Big(Q(s,a)+\underbrace{c_{\text{puct}}\,P(s,a)\tfrac{\sqrt{\sum_b N(s,b)}}{1+N(s,a)}}_{U(s,a)}\Big);
      \]
    \item \T{expand a leaf and evaluate it with the network $(P,v)=f_\theta(s)$;}%
          {espansione di una foglia e sua valutazione con la rete $(P,v)=f_\theta(s)$;}
    \item \T{bootstrap from the value $v$;}{bootstrap dal valore $v$;}
    \item \T{back up:}{propagazione all'indietro:} $N\!\mathrel{+}\!=\!1$, $W\!\mathrel{+}\!=\!v$, $Q=W/N$.
  \end{enumerate}
  \T{The visit counts give an improved policy,}{I conteggi delle visite danno una politica
  migliore,} $\pi(a\mid s)\propto N(s,a)^{1/\tau}$.
\end{frame}

\begin{frame}{\T{AlphaZero: self-play policy iteration}{AlphaZero: policy iteration con self-play}}
  \T{A single network $(p,v)=f_\theta(s)$ gives a policy $p$ over the moves and a value
  $v\in[-1,1]$ (how likely the search is to succeed from $s$).}%
  {Una sola rete $(p,v)=f_\theta(s)$ d\`a una politica $p$ sulle mosse e un valore
  $v\in[-1,1]$ (quanto \`e probabile che la ricerca riesca da $s$).}
  \begin{itemize}
    \item \T{Self-play: MCTS, guided by $f_\theta$, gives a stronger policy $\pi$, played to
          the end to obtain an outcome $z$.}%
          {Self-play: MCTS, guidata da $f_\theta$, d\`a una politica pi\`u forte $\pi$, giocata
          fino alla fine per ottenere un esito $z$.}
    \item \T{Learning: $v$ is regressed on $z$ and $p$ on $\pi$; MCTS improves the policy and
          the network evaluates it:}%
          {Apprendimento: $v$ \`e regredito su $z$ e $p$ su $\pi$; MCTS migliora la politica e la
          rete la valuta:}
      \[
        \ell=\underbrace{(z-v)^2}_{\T{value}{valore}}\;\underbrace{-\,\pi^{\!\top}\!\log p}_{\T{policy (cross-entropy)}{politica (cross-entropy)}}\;+\;\underbrace{c\lVert\theta\rVert^2}_{\T{regularization}{regolarizzazione}} .
      \]
    \item \T{No human data and no hand-made heuristic: the heuristic is learned from
          self-play alone.}%
          {Niente dati umani e nessuna euristica scritta a mano: l'euristica si impara dal solo
          self-play.}
  \end{itemize}
\end{frame}

\begin{frame}{\T{AlphaZeroSAT: the model}{AlphaZeroSAT: il modello}}
  \begin{columns}[t]
    \begin{column}{0.56\textwidth}
      \footnotesize
      \T{State: the CNF as a fixed-size clauses $\times$ variables matrix (entries in
      $\{+1,0,-1\}$ by polarity, as image channels).}%
      {Stato: la CNF come matrice clausole $\times$ variabili di dimensione fissa (valori in
      $\{+1,0,-1\}$ secondo la polarit\`a, come canali di un'immagine).}\\[1mm]
      \T{Network: a convolutional trunk ($32@8{\times}8$, $64@4{\times}4$, $64@3{\times}3$),
      then two heads:}{Rete: un tronco convoluzionale ($32@8{\times}8$, $64@4{\times}4$,
      $64@3{\times}3$), poi due teste:}
      \begin{align*}
        \T{policy}{politica}\ p &= \text{softmax}\big(\text{logits} + (\text{mask}-1)\!\cdot\!\infty\big)\in\Delta^{2n}\\
        \T{value}{valore}\ v &= \tanh(\cdot)\in[-1,1]
      \end{align*}
      \T{the literals of the variables absent from the remaining clauses are masked.}%
      {i letterali delle variabili assenti dalle clausole rimaste sono mascherati.}\\[1mm]
      \T{Model1/2/3 differ in the heads (fully connected or convolutional), as in the
      original design.}{Model1/2/3 differiscono nelle teste (completamente connesse o
      convoluzionali), come nel progetto originale.}
    \end{column}
    \begin{column}{0.44\textwidth}
      \footnotesize
      \T{Environment: MiniSat extended into an MCTS game that supports hypothetical
      simulations.}{Ambiente: MiniSat esteso in un gioco per MCTS che supporta simulazioni
      ipotetiche.}\\[2mm]
      \T{Measure: the mean number of branching decisions to solve (lower is better), as in
      the original paper.}{Misura: il numero medio di decisioni di branching per risolvere
      (meno \`e meglio), come nell'articolo originale.}\\[1mm]
      \T{Result (\texttt{uf20-91}, 25 cycles of self-play): $6.70$ mean decisions on 64 test
      problems; the evaluation after each cycle, on 16 problems, ranges from $3.4$ to $9.4$,
      too noisy to show a trend.}{Risultato (\texttt{uf20-91}, 25 cicli di self-play): $6.70$
      decisioni in media su 64 problemi di test; la valutazione dopo ogni ciclo, su 16
      problemi, va da $3.4$ a $9.4$, troppo rumorosa per mostrare una tendenza.}\\[1mm]
      \T{Each decision is an MCTS search, much slower than a decision of VSIDS.}%
      {Ogni decisione \`e una ricerca MCTS, molto pi\`u lenta di una decisione di VSIDS.}\\[2mm]
      \T{Limitation: with a fixed-size matrix the network cannot be applied to other sizes,
      which is the motivation of the graph methods.}{Limite: con una matrice di dimensione
      fissa la rete non si applica ad altre dimensioni, ed \`e la ragione dei metodi su grafo.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{\T{Restricted heuristics: fewer forward passes}{Euristiche ristrette: meno forward pass}}
  \T{Graph-Q-SAT reduces the iterations, but a forward pass of the network at each decision
  costs more than VSIDS. Three ways of querying the network less (Shirokikh et al., 2023),
  at evaluation time:}%
  {Graph-Q-SAT riduce le iterazioni, ma un forward pass della rete a ogni decisione costa pi\`u
  di VSIDS. Tre modi di interrogare meno la rete (Shirokikh et al., 2023), in valutazione:}
  \begin{itemize}
    \item \T{early release (\texttt{--release\_after T}): the model takes the first $T$
          decisions, VSIDS the others;}%
          {rilascio anticipato (\texttt{--release\_after T}): il modello prende le prime $T$
          decisioni, VSIDS le altre;}
    \item \T{action pool (\texttt{--action\_pool\_size L}): one forward pass gives the $L$
          best actions, executed in order;}%
          {pool di azioni (\texttt{--action\_pool\_size L}): un forward pass d\`a le $L$ azioni
          migliori, eseguite in ordine;}
    \item \T{warm start (\texttt{--warmstart\_release}): the $Q$-values of the root
          initialize the activities of VSIDS,}%
          {warm start (\texttt{--warmstart\_release}): i $Q$-value della radice
          inizializzano le attivit\`a di VSIDS,}
          $\text{activity}(x)=-1/\max\big(Q(x,\text{T}),Q(x,\text{F})\big)$,
          \T{which then runs unaided.}{che poi lavora da sola.}
  \end{itemize}
  \T{On the instances of this project the search is too short for the saving to show, and
  they are not measured.}{Sulle istanze di questo progetto la ricerca \`e troppo breve perch\'e
  il risparmio si veda, e non sono misurate.}
\end{frame}
